                    break;

                    var img = picSteve.Image;
                    img.RotateFlip(RotateFlipType.Rotate180FlipNone);
                    picSteve.Image = img;
                    picSteve.Refresh();
                    break;
                case 11:
                    Bitmap bmp = new Bitmap(picSteve.Image);
                    for (int y = 0; y < bmp.Height; y++)
                        for (int x = 0; x < bmp.Width; x++)
                            bmp.SetPixel(x, y, Color.FromArgb(255 - bmp.GetPixel(x, y).R,

                    picSteve.Image = bmp;
                    break;
            }
            intTime = 0;
            timReset.Start();
        }
